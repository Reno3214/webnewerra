    <script>
        // Snake Game
        const canvas = document.getElementById('snakeCanvas');
        const ctx = canvas.getContext('2d');
        const snakeScoreEl = document.getElementById('snakeScore');
        const snakeDirEl = document.getElementById('snakeDir');
        const snakeStatusEl = document.getElementById('snakeStatus');
        const resetSnakeBtn = document.getElementById('resetSnake');

        let gridSize = 20;
        let tileSize = 20;
        let snake = [];
        let food = {};
        let direction = '→';
        let gameLoop = null;
        let gameOver = false;

        // Initialize snake
        function initSnake() {
            snake = [
                { x: 10, y: 10 },
                { x: 9, y: 10 },
                { x: 8, y: 10 }
            ];
            snakeScoreEl.textContent = 0;
            updateSnakeStatus();
        }

        // Draw snake
        function drawSnake() {
            ctx.fillStyle = '#00ff88';
            snake.forEach((segment, index) => {
                ctx.fillRect(
                    (segment.x * tileSize) + 1,
                    (segment.y * tileSize) + 1,
                    tileSize - 2,
                    tileSize - 2
                );
            });
        }

        // Draw food
        function drawFood() {
            const x = Math.floor(Math.random() * gridSize);
            const y = Math.floor(Math.random() * gridSize);
            food = { x, y };
            ctx.fillStyle = '#ff4444';
            ctx.fillRect(
                (x * tileSize) + 1,
                (y * tileSize) + 1,
                tileSize - 2,
                tileSize - 2
            );
        }

        // Update snake status
        function updateSnakeStatus() {
            snakeStatusEl.textContent = direction === '→' ? '→' : '↓' || '↑' || '←' || '↓';
        }

        // Main game loop
        function gameLoop() {
            if (gameOver) return;

            // Move snake
            const head = { ...snake[0] };
            switch (direction) {
                case '→': head.x++; break;
                case '←': head.x--; break;
                case '↓': head.y++; break;
                case '↑': head.y--; break;
            }

            // Check wall collision
            if (head.x < 0 || head.x >= gridSize || head.y < 0 || head.y >= gridSize) {
                gameOver = true;
                return;
            }

            // Check self collision
            if (snake.some(seg => seg.x === head.x && seg.y === head.y)) {
                gameOver = true;
                return;
            }

            // Add new head
            snake.unshift(head);

            // Check food collision
            if (head.x === food.x && head.y === food.y) {
                // Increase score
                snakeScoreEl.textContent = parseInt(snakeScoreEl.textContent) + 1;
                updateSnakeStatus();
                
                // Spawn new food
                drawFood();
                // Don't remove tail - snake grows
            } else {
                snake.pop();
            }

            drawSnake();
            drawFood();

            gameLoop(); // Recursion for next frame
        }

        // Start game
        function startGame() {
            initSnake();
            document.getElementById('snakeScore').textContent = 0;
            document.getElementById('snakeDir').textContent = '→';
            document.getElementById('snakeStatus').textContent = 'Aktif';
            gameLoop();
        }

        // Reset game
        resetSnakeBtn.addEventListener('click', () => {
            initSnake();
            document.getElementById('snakeScore').textContent = 0;
            document.getElementById('snakeDir').textContent = '→';
            document.getElementById('snakeStatus').textContent = 'Aktif';
            gameLoop();
        });

        // Start when page loads
        window.onload = startGame;

        // Bot Battle Calculation
        const botScores = {
            A: 0,
            B: 0
        };

        function calculateBotPerformance() {
            // Simulate bot performance calculation
            const totalBattles = 100;
            const botAWins = Math.floor(totalBattles * 0.45);
            const botBWins = totalBattles - botAWins;

            botAPerf.textContent = 'Score: ' + botAWins;
            botBPerf.textContent = 'Score: ' + botBWins;

            // Determine winner
            if (botAWins > botBWins) {
                document.getElementById('finalScore').textContent = `${botAWins}-${botBWins}`;
                document.getElementById('botAPerf').textContent = '胜利!';
                document.getElementById('botBPerf').textContent = 'Pahit!';
            } else if (botBWins > botAWins) {
                document.getElementById('finalScore').textContent = `${botBWins}-${botAWins}`;
                document.getElementById('finalScore').className = 'text-center p-6 bg-[#1a1a2e] rounded-lg neon-text';
                document.getElementById('botAPerf').textContent = 'Pahit!';
                document.getElementById('botBPerf').textContent = '胜利!';
            } else {
                document.getElementById('finalScore').textContent = '50-50';
                document.getElementById('botAPerf').textContent = 'Dempun';
                document.getElementById('botBPerf').textContent = 'Dempun';
            }
        }

        // Initial bot calculation
        calculateBotPerformance();

        // Add some animations
        setInterval(calculateBotPerformance, 3000);
    </script>
